async def resume_session(session_id: str) -> SharedContext:
    ctx = await memory.load_context(session_id)
    completed = [t for t in ctx.task_dag if t.status == "completed"]
    pending = [t for t in ctx.task_dag if t.status in ["pending", "ready", "running"]]
    
    # Tasks that were "running" when crash happened need to be reset to "pending"
