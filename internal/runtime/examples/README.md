## 3. Go Binding Usage Example

A Go code snippet demonstrating how to generate a signed Event Envelope and publish it to the event bus:

```go
package main

import (
	"context"
	"time"
	
	"github.com/fallofpheonix/PhoenixCore/bus"
	v1 "github.com/fallofpheonix/PhoenixCore/proto/v1"
	"google.golang.org/protobuf/types/known/timestamppb"
)

func main() {
	// 1. Construct the event envelope
	evt := &v1.EventEnvelope{
		EventId:        "evt_example_001",
		EventVersion:   "1.0.0",
		EventType:      "phoenix.example.action",
		SourceRepo:     "PhoenixKernel",
		SourceComponent: "example_runner",
		ReplaySequence: 1,
		SchemaVersion:  "1.0.0",
		CreatedAt:      timestamppb.New(time.Now()),
		UpdatedAt:      timestamppb.New(time.Now()),
	}

	// 2. Cryptographically sign the envelope payload
	err := bus.SignEnvelope(evt, privateKey)
	if err != nil {
		panic(err)
	}

	// 3. Publish to event bus
	err = bus.Publish(context.Background(), "phoenix.events.normal", evt)
	if err != nil {
		panic(err)
	}
}
```
